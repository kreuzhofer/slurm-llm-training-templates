# Pinned stack for LTX-2.5 video generation on NVIDIA B300 (sm_103), installed
# by models/ltx-2.5/setup.sh into its own venv, $TEMPLATES_DIR/venv-ltx.
#
# Why not the shared venv: ltx-core requires transformers<5.15. 5.15 routes
# config attribute access through a layer that raises on the global
# `config.head_dim` its own Gemma 4 code still reads, so the text encoder cannot
# be built. The shared venv pins 5.16.1 for vLLM 0.31.0.
#
# Why cu130 and not upstream's cu132: the driver here is 580.159.04, which is
# CUDA 13.0. torch 2.13.0+cu130 is the build the shared venv already runs on
# B300, and natten publishes a matching +torch2130cu130 wheel.

--extra-index-url https://download.pytorch.org/whl/cu130
--find-links https://whl.natten.org

torch==2.13.0+cu130
torchvision==0.28.0+cu130
# There is no torchaudio 2.13 on any index. 2.11 ships stable-ABI (abi3)
# libraries and declares no torch requirement; upstream LTX also runs 2.11 with
# torch 2.13. ltx-core uses it for a mel spectrogram and resampling only.
torchaudio==2.11.0+cu130

# Neighbourhood attention for the diffusion video decoder (vae/ltx-2.5-video-vae).
# Without it the decoder falls back to Triton, which upstream calls slower.
natten==0.21.7+torch2130cu130

transformers==5.14.1
accelerate==1.15.0

# Build-time only: ltx-kernels compiles CUDA extensions against torch, and
# torch refuses an nvcc whose major version differs from its own. The system
# toolkit on these nodes is 12.9, so nvcc comes from pip and matches cu130.
# nvidia-cuda-nvcc leaves its compiler front end (nvvm) and headers (crt)
# unpinned, and pip then installs 13.4 under a 13.0 nvcc. The build then fails
# with "ptxas fatal: Unsupported .version 9.4; current version is '9.0'".
nvidia-cuda-nvcc==13.0.88
nvidia-cuda-crt==13.0.88
nvidia-nvvm==13.0.88
nvidia-cuda-cccl==13.0.85
# Without ninja, torch compiles the kernel sources one at a time.
ninja==1.13.0

# ltx-kernels runtime: CuTe DSL kernels for the diffusion VAE on datacenter
# Blackwell.
nvidia-cutlass-dsl==4.8.0
